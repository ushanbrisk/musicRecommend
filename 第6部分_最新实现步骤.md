### 1.3 生成歌曲向量数据

**目标**：为所有歌曲生成 Embedding 向量并写入 song_embeddings 表

#### 步骤1：创建向量生成脚本

创建文件 `/home/luke/code_project/musicRecommend/scripts/generate_embeddings.py`

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
歌曲 Embedding 向量生成脚本
功能：从 songs 表读取数据，生成向量，写入 song_embeddings 表
"""

import os
import sys
import psycopg2
from openai import OpenAI
from dotenv import load_dotenv
import time

# 加载环境变量
backend_env = os.path.expanduser('~/code_project/music-project/backend/.env')
load_dotenv(backend_env)

# 初始化 Embedding 客户端
client = OpenAI(
    api_key=os.getenv('EMBEDDING_API_KEY'),
    base_url=os.getenv('EMBEDDING_PROVIDER_URL', 'https://api.minimax.chat')
)
EMBEDDING_MODEL = os.getenv('EMBEDDING_MODEL_NAME', 'embo-01')

# 数据库连接
def get_db_connection():
    return psycopg2.connect(
        host=os.getenv('POSTGRES_HOST', 'localhost'),
        port=os.getenv('POSTGRES_PORT', 5432),
        database=os.getenv('POSTGRES_DB', 'musicdb'),
        user=os.getenv('POSTGRES_USER', 'postgres'),
        password=os.getenv('POSTGRES_PASSWORD', '')
    )

def generate_embedding(text: str):
    """调用 Embedding API 生成向量"""
    try:
        response = client.embeddings.create(
            model=EMBEDDING_MODEL,
            input=text
        )
        return response.data[0].embedding
    except Exception as e:
        print(f"Error generating embedding: {e}")
        return None

def get_songs_without_embeddings(conn, limit=1000):
    """获取尚未生成向量的歌曲"""
    cursor = conn.cursor()
    query = """
        SELECT
            s.song_id,
            s.song_name,
            s.artist,
            s.album,
            spa.playlist_names,
            spa.playlist_categories
        FROM songs s
        LEFT JOIN song_embeddings se ON s.song_id = se.song_id
        LEFT JOIN song_playlist_agg spa ON s.song_id = spa.song_id
        WHERE se.id IS NULL
        LIMIT %s
    """
    cursor.execute(query, (limit,))
    rows = cursor.fetchall()
    cursor.close()

    songs = []
    for row in rows:
        songs.append({
            'song_id': row[0],
            'song_name': row[1],
            'artist': row[2],
            'album': row[3],
            'playlist_names': row[4] or '',
            'playlist_categories': row[5] or ''
        })
    return songs

def build_text_description(song):
    """构建歌曲的文本描述（用于生成 embedding）"""
    parts = []

    # 歌曲名
    if song['song_name']:
        parts.append(song['song_name'])

    # 艺术家
    if song['artist']:
        parts.append(song['artist'])

    # 专辑
    if song['album']:
        parts.append(song['album'])

    # 歌单名称（最多取5个）
    if song['playlist_names']:
        playlists = song['playlist_names'].split(',')[:5]
        parts.extend(playlists)

    # 歌单分类
    if song['playlist_categories']:
        categories = song['playlist_categories'].split(',')[:3]
        parts.extend(categories)

    return ' '.join(parts)

def save_embedding(conn, song_id, text_description, embedding_vector):
    """保存向量到数据库"""
    cursor = conn.cursor()
    query = """
        INSERT INTO song_embeddings (
            song_id, text_description, embedding
        ) VALUES (%s, %s, %s)
        ON CONFLICT (song_id) DO UPDATE SET
            text_description = EXCLUDED.text_description,
            embedding = EXCLUDED.embedding,
            updated_at = CURRENT_TIMESTAMP
    """
    cursor.execute(query, (song_id, text_description, embedding_vector))
    conn.commit()
    cursor.close()

def main():
    print("开始生成歌曲 Embedding 向量...")

    conn = get_db_connection()

    # 统计总数
    cursor = conn.cursor()
    cursor.execute("SELECT COUNT(*) FROM songs")
    total_songs = cursor.fetchone()[0]
    cursor.execute("SELECT COUNT(*) FROM song_embeddings")
    existing_count = cursor.fetchone()[0]
    cursor.close()

    print(f"歌曲总数: {total_songs}")
    print(f"已生成向量: {existing_count}")
    print(f"待生成: {total_songs - existing_count}")

    batch_size = 1000
    processed = 0

    while True:
        songs = get_songs_without_embeddings(conn, limit=batch_size)
        if not songs:
            print("所有歌曲向量已生成完毕！")
            break

        print(f"\n批次处理: {len(songs)} 首歌曲")

        for i, song in enumerate(songs, 1):
            song_id = song['song_id']
            text_desc = build_text_description(song)

            # 生成向量
            embedding = generate_embedding(text_desc)
            if embedding is None:
                print(f"  [{i}/{len(songs)}] Song {song_id}: 向量生成失败，跳过")
                continue

            # 保存到数据库
            save_embedding(conn, song_id, text_desc, embedding)
            processed += 1

            if i % 100 == 0:
                print(f"  [{i}/{len(songs)}] 已处理 {processed} 首")

            # 避免 API 限流
            time.sleep(0.05)

        print(f"批次完成，累计处理: {processed} 首")

    conn.close()
    print(f"\n总计生成向量: {processed} 首")

if __name__ == '__main__':
    main()
```

## 阶段3：推荐服务实现（预计工作量：2-3天）

### 3.1 实现 RecommendService 的两阶段推荐

**目标**：实现向量召回 + LLM精排的完整流程

编辑 `/home/luke/code_project/music-project/backend/services/recommend_service.py`

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
推荐服务：两阶段推荐算法
- 阶段1：向量数据库粗召回（top 200）
- 阶段2：LLM 精排序（top 20）
"""

import psycopg2
from typing import List, Dict, Optional
import json
import time
from .llm_service import LLMService

class RecommendService:
    def __init__(self, db_connection):
        self.db = db_connection
        self.llm = LLMService()

    def recommend(
        self,
        query: str,
        session_id: Optional[str] = None,
        user_id: Optional[str] = None,
        max_results: int = 20
    ) -> Dict:
        """
        两阶段推荐主流程
        """
        start_time = time.time()

        # 阶段1：向量召回
        candidates = self._vector_recall(query, recall_size=200)

        if not candidates:
            return {
                "success": True,
                "query": query,
                "results": [],
                "total": 0,
                "latency_ms": int((time.time() - start_time) * 1000),
                "message": "未找到匹配的歌曲"
            }

        # 阶段2：LLM 精排
        ranked_results = self._llm_rerank(query, candidates, top_k=max_results)

        # 记录历史
        history_id = self._save_history(
            session_id=session_id,
            user_id=user_id,
            query_text=query,
            result_count=len(ranked_results),
            latency_ms=int((time.time() - start_time) * 1000)
        )

        return {
            "success": True,
            "query": query,
            "history_id": history_id,
            "results": ranked_results,
            "total": len(ranked_results),
            "latency_ms": int((time.time() - start_time) * 1000)
        }

    def _vector_recall(self, query: str, recall_size: int = 200) -> List[Dict]:
        """
        阶段1：向量召回
        使用 pgvector 进行相似度检索
        """
        # 1. 生成查询向量
        query_embedding = self.llm.generate_embedding(query)
        if not query_embedding:
            print("查询向量生成失败")
            return []

        # 2. 向量检索
        cursor = self.db.cursor()

        # 将向量转为字符串格式
        embedding_str = '[' + ','.join(map(str, query_embedding)) + ']'

        query_sql = """
            SELECT
                se.song_id,
                s.song_name,
                s.artist,
                s.album,
                se.text_description,
                se.genre,
                se.mood,
                se.scene,
                1 - (se.embedding <=> %s::vector) AS similarity
            FROM song_embeddings se
            JOIN songs s ON se.song_id = s.song_id
            WHERE se.embedding IS NOT NULL
            ORDER BY se.embedding <=> %s::vector
            LIMIT %s
        """

        cursor.execute(query_sql, (embedding_str, embedding_str, recall_size))
        rows = cursor.fetchall()
        cursor.close()

        candidates = []
        for row in rows:
            candidates.append({
                'song_id': row[0],
                'song_name': row[1],
                'artist': row[2],
                'album': row[3],
                'text_description': row[4],
                'genre': row[5],
                'mood': row[6],
                'scene': row[7],
                'similarity': float(row[8])
            })

        print(f"向量召回: {len(candidates)} 首候选歌曲")
        return candidates

    def _llm_rerank(self, query: str, candidates: List[Dict], top_k: int = 20) -> List[Dict]:
        """
        阶段2：LLM 精排序
        """
        # 构建候选歌曲上下文
        songs_context = self._build_songs_context(candidates)

        # 构建 LLM Prompt
        prompt = f"""用户查询："{query}"

候选歌曲（{len(candidates)}首）：
{songs_context}

请从候选歌曲中选择最匹配用户查询的歌曲，返回 JSON 格式：
{{
  "matched_songs": [
    {{
      "song_id": 123,
      "match_score": 0.95,
      "match_reason": "匹配原因说明"
    }}
  ]
}}

要求：
1. 理解用户查询中的所有要素（风格、情绪、乐器、场景等）
2. 选择最符合用户需求的歌曲
3. match_score 范围：0.6-1.0，低于0.6的不返回
4. 最多返回 {top_k} 首歌曲
5. 提供简洁的匹配理由（20字以内）
"""

        # 调用 LLM
        try:
            response = self.llm.chat([{"role": "user", "content": prompt}])
            result = json.loads(response)
            matched_songs = result.get('matched_songs', [])

            # 补充完整信息
            final_results = []
            for matched in matched_songs:
                song_id = matched['song_id']
                # 从候选中找到完整信息
                for candidate in candidates:
                    if candidate['song_id'] == song_id:
                        final_results.append({
                            'song_id': song_id,
                            'song_name': candidate['song_name'],
                            'artist': candidate['artist'],
                            'album': candidate['album'],
                            'match_score': matched['match_score'],
                            'match_reason': matched.get('match_reason', '')
                        })
                        break

            print(f"LLM 精排: {len(final_results)} 首最终结果")
            return final_results

        except Exception as e:
            print(f"LLM 精排失败: {e}")
            # 降级：返回向量召回的 top_k
            return candidates[:top_k]

    def _build_songs_context(self, candidates: List[Dict]) -> str:
        """构建候选歌曲的上下文文本"""
        context_lines = []
        for i, song in enumerate(candidates, 1):
            line = f"{i}. 歌名：{song['song_name']}, 艺术家：{song['artist']}"
            if song.get('genre'):
                line += f", 风格：{song['genre']}"
            if song.get('mood'):
                line += f", 情绪：{song['mood']}"
            if song.get('scene'):
                line += f", 场景：{song['scene']}"
            context_lines.append(line)

        return '\n'.join(context_lines)

    def _save_history(
        self,
        session_id: Optional[str],
        user_id: Optional[str],
        query_text: str,
        result_count: int,
        latency_ms: int
    ) -> int:
        """保存推荐历史"""
        cursor = self.db.cursor()
        query = """
            INSERT INTO recommendation_history (
                session_id, user_id, query_text, query_type,
                result_count, latency_ms
            ) VALUES (%s, %s, %s, %s, %s, %s)
            RETURNING id
        """
        cursor.execute(query, (
            session_id, user_id, query_text, 'complex',
            result_count, latency_ms
        ))
        history_id = cursor.fetchone()[0]
        self.db.commit()
        cursor.close()
        return history_id
```

### 4.2 性能基准测试

```python
#!/usr/bin/env python3
import requests
import time
import statistics

def benchmark_recommend():
    """性能基准测试"""
    test_queries = [
        "欢快的凯尔特音乐，有小提琴伴奏",
        "适合工作的轻音乐",
        "悲伤的钢琴曲",
        "激昂的摇滚音乐"
    ]

    latencies = []

    for query in test_queries:
        start = time.time()
        response = requests.post(
            'http://localhost:5000/api/recommend',
            json={'query': query, 'session_id': 'benchmark'}
        )
        latency = (time.time() - start) * 1000
        latencies.append(latency)

        result = response.json()
        print(f"查询: {query}")
        print(f"  响应时间: {latency:.0f} ms")
        print(f"  结果数量: {result.get('total', 0)}\n")

    print("=" * 60)
    print("性能统计:")
    print(f"  平均响应时间: {statistics.mean(latencies):.0f} ms")
    print(f"  最小响应时间: {min(latencies):.0f} ms")
    print(f"  最大响应时间: {max(latencies):.0f} ms")
    print(f"  标准差: {statistics.stdev(latencies):.0f} ms")

if __name__ == '__main__':
    benchmark_recommend()
```

**性能目标**：
- 向量召回：< 100 ms
- LLM 精排：< 2000 ms
- 总响应时间：< 2500 ms

### 4.3 结果质量验证

创建质量验证脚本：

```python
#!/usr/bin/env python3
import requests

def validate_quality():
    """验证推荐结果质量"""
    test_cases = [
        {
            'query': '欢快的凯尔特音乐，有小提琴伴奏',
            'expected_keywords': ['凯尔特', 'Celtic', '小提琴', 'violin', '欢快'],
            'description': '复杂查询：多个条件组合'
        },
        {
            'query': '适合工作的背景音乐',
            'expected_keywords': ['工作', '轻音乐', '背景', '轻快', '舒缓'],
            'description': '场景查询：工作场景'
        },
        {
            'query': '悲伤的钢琴曲',
            'expected_keywords': ['钢琴', 'piano', '悲伤', '忧伤', '伤感'],
            'description': '情绪+乐器查询'
        }
    ]

    for test in test_cases:
        print(f"\n{'='*60}")
        print(f"测试: {test['description']}")
        print(f"查询: {test['query']}")
        print(f"{'='*60}")

        response = requests.post(
            'http://localhost:5000/api/recommend',
            json={'query': test['query'], 'session_id': 'quality_test'}
        )

        result = response.json()

        if not result.get('results'):
            print("❌ 无结果返回")
            continue

        # 检查前5个结果
        matched_keywords = set()
        for song in result['results'][:5]:
            song_text = (
                song.get('song_name', '') + ' ' +
                song.get('artist', '') + ' ' +
                song.get('match_reason', '')
            ).lower()

            for keyword in test['expected_keywords']:
                if keyword.lower() in song_text:
                    matched_keywords.add(keyword)

        coverage = len(matched_keywords) / len(test['expected_keywords']) * 100
        print(f"\n关键词覆盖率: {coverage:.0f}%")
        print(f"匹配关键词: {matched_keywords}")
        print(f"期望关键词: {set(test['expected_keywords'])}")

        if coverage >= 40:  # 至少40%的关键词匹配
            print("✅ 质量合格")
        else:
            print("⚠️ 质量有待提升")

if __name__ == '__main__':
    validate_quality()
```
