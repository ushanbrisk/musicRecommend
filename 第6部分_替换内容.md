#### 6.1.3 生成歌曲向量数据

**重要文件**：`/home/luke/code_project/musicRecommend/scripts/generate_embeddings.py`

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
歌曲 Embedding 向量生成脚本
"""
import os
import psycopg2
from openai import OpenAI
from dotenv import load_dotenv
import time

# 加载环境变量
backend_env = os.path.expanduser('~/code_project/music-project/backend/.env')
load_dotenv(backend_env)

# 初始化 Embedding 客户端
client = OpenAI(
    api_key=os.getenv('EMBEDDING_API_KEY'),
    base_url=os.getenv('EMBEDDING_PROVIDER_URL', 'https://api.minimax.chat')
)
EMBEDDING_MODEL = os.getenv('EMBEDDING_MODEL_NAME', 'embo-01')

def get_db_connection():
    return psycopg2.connect(
        host=os.getenv('POSTGRES_HOST', 'localhost'),
        port=os.getenv('POSTGRES_PORT', 5432),
        database=os.getenv('POSTGRES_DB', 'musicdb'),
        user=os.getenv('POSTGRES_USER', 'postgres'),
        password=os.getenv('POSTGRES_PASSWORD', '')
    )

def generate_embedding(text: str):
    try:
        response = client.embeddings.create(model=EMBEDDING_MODEL, input=text)
        return response.data[0].embedding
    except Exception as e:
        print(f"Error: {e}")
        return None

def get_songs_without_embeddings(conn, limit=1000):
    cursor = conn.cursor()
    query = """
        SELECT s.song_id, s.song_name, s.artist, s.album,
               spa.playlist_names, spa.playlist_categories
        FROM songs s
        LEFT JOIN song_embeddings se ON s.song_id = se.song_id
        LEFT JOIN song_playlist_agg spa ON s.song_id = spa.song_id
        WHERE se.id IS NULL
        LIMIT %s
    """
    cursor.execute(query, (limit,))
    return cursor.fetchall()

def build_text_description(row):
    """构建文本描述：歌名 + 艺术家 + 专辑 + 歌单名"""
    parts = [row[1], row[2], row[3]]  # song_name, artist, album
    if row[4]:  # playlist_names
        parts.extend(row[4].split(',')[:5])
    return ' '.join(filter(None, parts))

def save_embedding(conn, song_id, text_desc, embedding):
    cursor = conn.cursor()
    query = """
        INSERT INTO song_embeddings (song_id, text_description, embedding)
        VALUES (%s, %s, %s)
        ON CONFLICT (song_id) DO UPDATE SET
            text_description = EXCLUDED.text_description,
            embedding = EXCLUDED.embedding,
            updated_at = CURRENT_TIMESTAMP
    """
    cursor.execute(query, (song_id, text_desc, embedding))
    conn.commit()

def main():
    print("开始生成歌曲 Embedding 向量...")
    conn = get_db_connection()

    processed = 0
    batch_size = 1000

    while True:
        rows = get_songs_without_embeddings(conn, limit=batch_size)
        if not rows:
            print("所有歌曲向量已生成完毕！")
            break

        print(f"\n批次处理: {len(rows)} 首")
        for i, row in enumerate(rows, 1):
            song_id = row[0]
            text_desc = build_text_description(row)
            embedding = generate_embedding(text_desc)

            if embedding:
                save_embedding(conn, song_id, text_desc, embedding)
                processed += 1

            if i % 100 == 0:
                print(f"  已处理 {i}/{len(rows)}")
            time.sleep(0.05)  # 避免API限流

    conn.close()
    print(f"\n总计生成: {processed} 首")

if __name__ == '__main__':
    main()
```

### 6.3 阶段3：推荐服务实现（预计工作量：2-3天）★★★

#### 6.3.1 实现 RecommendService 的两阶段推荐

编辑 `backend/services/recommend_service.py`：

```python
#!/usr/bin/env python3
"""
两阶段推荐服务
- 阶段1：向量召回（top 200）
- 阶段2：LLM 精排（top 20）
"""
import psycopg2
import json
import time
from .llm_service import LLMService

class RecommendService:
    def __init__(self, db_connection):
        self.db = db_connection
        self.llm = LLMService()

    def recommend(self, query, session_id=None, user_id=None, max_results=20):
        """主推荐流程"""
        start_time = time.time()

        # 阶段1：向量召回
        candidates = self._vector_recall(query, recall_size=200)

        if not candidates:
            return {
                "success": True,
                "query": query,
                "results": [],
                "total": 0,
                "latency_ms": int((time.time() - start_time) * 1000)
            }

        # 阶段2：LLM 精排
        results = self._llm_rerank(query, candidates, top_k=max_results)

        # 保存历史
        history_id = self._save_history(
            session_id, user_id, query, len(results),
            int((time.time() - start_time) * 1000)
        )

        return {
            "success": True,
            "query": query,
            "history_id": history_id,
            "results": results,
            "total": len(results),
            "latency_ms": int((time.time() - start_time) * 1000)
        }

    def _vector_recall(self, query, recall_size=200):
        """阶段1：向量召回"""
        # 1. 生成查询向量
        query_embedding = self.llm.generate_embedding(query)
        if not query_embedding:
            return []

        # 2. 向量检索
        cursor = self.db.cursor()
        embedding_str = '[' + ','.join(map(str, query_embedding)) + ']'

        query_sql = """
            SELECT se.song_id, s.song_name, s.artist, s.album,
                   se.genre, se.mood, se.scene,
                   1 - (se.embedding <=> %s::vector) AS similarity
            FROM song_embeddings se
            JOIN songs s ON se.song_id = s.song_id
            WHERE se.embedding IS NOT NULL
            ORDER BY se.embedding <=> %s::vector
            LIMIT %s
        """

        cursor.execute(query_sql, (embedding_str, embedding_str, recall_size))
        rows = cursor.fetchall()
        cursor.close()

        candidates = []
        for row in rows:
            candidates.append({
                'song_id': row[0],
                'song_name': row[1],
                'artist': row[2],
                'album': row[3],
                'genre': row[4],
                'mood': row[5],
                'scene': row[6],
                'similarity': float(row[7])
            })

        print(f"向量召回: {len(candidates)} 首候选")
        return candidates

    def _llm_rerank(self, query, candidates, top_k=20):
        """阶段2：LLM 精排序"""
        # 构建候选上下文
        context = '\n'.join([
            f"{i+1}. {s['song_name']} - {s['artist']} "
            f"[风格:{s.get('genre','')} 情绪:{s.get('mood','')}]"
            for i, s in enumerate(candidates)
        ])

        prompt = f"""用户查询："{query}"

候选歌曲（{len(candidates)}首）：
{context}

请选择最匹配的歌曲，返回JSON：
{{
  "matched_songs": [
    {{"song_id": 123, "match_score": 0.95, "match_reason": "理由"}}
  ]
}}

要求：match_score >= 0.6，最多{top_k}首，理由20字内。
"""

        try:
            response = self.llm.chat([{"role": "user", "content": prompt}])
            result = json.loads(response)
            matched = result.get('matched_songs', [])

            # 补充完整信息
            final = []
            for m in matched:
                for c in candidates:
                    if c['song_id'] == m['song_id']:
                        final.append({
                            'song_id': c['song_id'],
                            'song_name': c['song_name'],
                            'artist': c['artist'],
                            'album': c['album'],
                            'match_score': m['match_score'],
                            'match_reason': m.get('match_reason', '')
                        })
                        break

            print(f"LLM 精排: {len(final)} 首最终结果")
            return final

        except Exception as e:
            print(f"LLM 精排失败: {e}，降级返回向量结果")
            return candidates[:top_k]

    def _save_history(self, session_id, user_id, query, count, latency):
        """保存推荐历史"""
        cursor = self.db.cursor()
        query_sql = """
            INSERT INTO recommendation_history
            (session_id, user_id, query_text, query_type, result_count, latency_ms)
            VALUES (%s, %s, %s, 'complex', %s, %s)
            RETURNING id
        """
        cursor.execute(query_sql, (session_id, user_id, query, count, latency))
        history_id = cursor.fetchone()[0]
        self.db.commit()
        cursor.close()
        return history_id
```
